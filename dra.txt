local function NewSlider(name, min, max, default, order, callback, parent)
    local frame = Instance.new("Frame", parent or Container)
    frame.Size = UDim2.new(1, -30, 0, 75); frame.BackgroundTransparency = 1; frame.LayoutOrder = order
    local lab = Instance.new("TextLabel", frame)
    lab.Size = UDim2.new(1, 0, 0, 25); lab.Text = name .. ": " .. default; lab.TextColor3 = Color3.new(1,1,1); lab.BackgroundTransparency = 1; lab.Font = Enum.Font.GothamBold; lab.TextSize = 14; lab.TextXAlignment = Enum.TextXAlignment.Left
    local back = Instance.new("Frame", frame)
    back.Size = UDim2.new(1, 0, 0, 12); back.Position = UDim2.new(0, 0, 0, 35); back.BackgroundColor3 = Color3.fromRGB(45, 45, 45); Instance.new("UICorner", back)
    local fill = Instance.new("Frame", back)
    fill.Size = UDim2.new((default - min) / (max - min), 0, 1, 0); fill.BackgroundColor3 = Color3.fromRGB(0, 170, 255); Instance.new("UICorner", fill)
    
    local sliding = false
    local function Upd()
        local mousePos = UserInputService:GetMouseLocation().X
        local pct = math.clamp((mousePos - back.AbsolutePosition.X) / back.AbsoluteSize.X, 0, 1)
        fill.Size = UDim2.new(pct, 0, 1, 0)
        local val = math.floor(min + (max - min) * pct)
        lab.Text = name .. ": " .. val; callback(val)
    end
    back.InputBegan:Connect(function(i) if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then sliding = true; Upd() end end)
    UserInputService.InputEnded:Connect(function(i) if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then sliding = false end end)
    RunService.RenderStepped:Connect(function() if sliding then Upd() end end)
    return frame
end

local function CreateList(name, color, order, btnText)
    local holder = Instance.new("Frame", Container)
    holder.Size = UDim2.new(1, 0, 0, 0); holder.BackgroundTransparency = 1; holder.ClipsDescendants = true; holder.LayoutOrder = order + 1
    local layout = Instance.new("UIListLayout", holder); layout.Padding = UDim.new(0, 8); layout.HorizontalAlignment = Enum.HorizontalAlignment.Center
    local mainBtn = NewButton(btnText, color, order, function(self)
        local isOpen = holder.Size.Y.Offset > 0
        local targetSize = isOpen and 0 or layout.AbsoluteContentSize.Y

        TweenService:Create(holder, TweenInfo.new(0.3), {Size = UDim2.new(1, 0, 0, targetSize)}):Play()
    end)
    return holder, mainBtn
end

-- LÓGICA ESP
local function CreateESP(p)
    local highlight = Instance.new("Highlight")
    highlight.Name = "ESP_" .. p.Name
    highlight.FillColor = Color3.fromRGB(255, 0, 0)
    highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
    highlight.FillTransparency = 0.5
    highlight.OutlineTransparency = 0
    
    local function Update()
        if _G.ESPEnabled and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            highlight.Adornee = p.Character
            highlight.Parent = ScreenGui
        else
            highlight.Adornee = nil
            highlight.Parent = nil
        end
    end
    RunService.RenderStepped:Connect(Update)
end

-- LÓGICA DE FLING E AUXILIARES
local function GetClosestPlayer()
    local closest, shortestDist = nil, math.huge
    local mousePos = UserInputService:GetMouseLocation()
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= Player and p.Character and p.Character:FindFirstChild("Head") then
            local pos, onScreen = Camera:WorldToViewportPoint(p.Character.Head.Position)
            if onScreen then
                local dist = (Vector2.new(pos.X, pos.Y) - mousePos).Magnitude
                if dist < shortestDist then closest = p; shortestDist = dist end
            end
        end
    end
    return closest
end

local function ExecuteFling(target)
    if States.Flinging or not target.Character then return end
    local char = Player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local tHRP = target.Character:FindFirstChild("HumanoidRootPart")
    if not hrp or not tHRP then return end
    States.Flinging = true
    local noclip = RunService.Stepped:Connect(function()
        for _, v in pairs(char:GetDescendants()) do if v:IsA("BasePart") then v.CanCollide = false end end
    end)
    local bav = Instance.new("BodyAngularVelocity", hrp)
    bav.AngularVelocity = Vector3.new(0, _G.FlingForce/10, 0); bav.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
    local start = tick()
    while tick() - start < 0.5 and States.Flinging do
        if tHRP then hrp.CFrame = tHRP.CFrame * CFrame.Angles(0, math.rad(tick()*720), 0) end
        task.wait()
    end
    bav:Destroy(); noclip:Disconnect(); hrp.AssemblyLinearVelocity = Vector3.zero; States.Flinging = false
end

local function Refresh()
    for _, h in pairs({InfH, TPH}) do for _, c in pairs(h:GetChildren()) do if c:IsA("TextButton") then c:Destroy() end end end
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= Player then
            local inf = NewButton(p.DisplayName, Color3.fromRGB(100, 0, 160), 1, function()
                States.InfiniteFling = true; InfiniteFlingTarget = p
                if InfiniteFlingThread then task.cancel(InfiniteFlingThread) end
                InfiniteFlingThread = task.spawn(function() while InfiniteFlingTarget == p do ExecuteFling(p); task.wait(0.1) end end)
            end)
            inf.Parent = InfH; inf.Size = UDim2.new(1, -40, 0, 35)
            
            local tp = NewButton("TP: "..p.DisplayName, Color3.fromRGB(25, 90, 170), 1, function()
                if p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                    Player.Character.HumanoidRootPart.CFrame = p.Character.HumanoidRootPart.CFrame * CFrame.new(0,0,3)
                end
            end)
            tp.Parent = TPH; tp.Size = UDim2.new(1, -40, 0, 35)
        end
    end
end

-- LOOPS DE ATUALIZAÇÃO
RunService.RenderStepped:Connect(function()
    local hum = Player.Character and Player.Character:FindFirstChild("Humanoid")
    if hum then 
        hum.WalkSpeed = _G.WalkSpeedValue 
        hum.JumpPower = _G.JumpPowerValue 
        hum.UseJumpPower = true 
    end
    
    local hrp = Player.Character and Player.Character:FindFirstChild("HumanoidRootPart")
    if States.Fly and hrp and hrp:FindFirstChild("FlyVel") then
        hrp.FlyGyro.CFrame = Camera.CFrame
        local dir = Vector3.zero
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir += Vector3.new(0,0,-1) end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir += Vector3.new(0,0,1) end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir += Vector3.new(-1,0,0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir += Vector3.new(1,0,0) end
        hrp.FlyVel.Velocity = Camera.CFrame:VectorToWorldSpace(dir * _G.FlySpeed)
    end
    
    if _G.AimbotEnabled and UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
        local target = GetClosestPlayer()
        if target and target.Character and target.Character:FindFirstChild("Head") then
            Camera.CFrame = Camera.CFrame:lerp(CFrame.new(Camera.CFrame.Position, target.Character.Head.Position), 1 - _G.AimbotSmoothness)
        end
    end
end)
