local function new(class, props)
    local inst = Instance.new(class)
    for k, v in pairs(props) do if k ~= "Parent" then inst[k] = v end end
    if props.Parent then inst.Parent = props.Parent end
    return inst
end
local function corner(inst, r) return new("UICorner", { CornerRadius = UDim.new(0, r), Parent = inst }) end
local function circle(inst)    return new("UICorner", { CornerRadius = UDim.new(1, 0), Parent = inst }) end
local function stroke(inst, color, thick)
    return new("UIStroke", { Color = color or Theme.Stroke, Thickness = thick or 1,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border, Parent = inst })
end
local function insetBorder(parent, radius)
    local f = new("Frame", { Size = UDim2.new(1,-2,1,-2), Position = UDim2.fromOffset(1,1),
        BackgroundTransparency = 1, ZIndex = 50, Parent = parent })
    corner(f, math.max(radius-1, 0))
    stroke(f, Theme.Stroke, 1)
    return f
end
local function tween(inst, t, props, style, dir)
    local tw = TweenService:Create(inst, TweenInfo.new(t, style or Enum.EasingStyle.Linear,
        dir or Enum.EasingDirection.Out), props)
    tw:Play()
    return tw
end
local function connect(sig, fn)
    local c = sig:Connect(fn)
    table.insert(Conns, c)
    return c
end
local function ord(parent)
    Orders[parent] = (Orders[parent] or 0) + 1
    return Orders[parent]
end

end

local function getPlayerRole(pl)
    local char = pl.Character
    local backpack = pl.Backpack
    
    if char then
        for _, item in ipairs(char:GetChildren()) do
            if item:IsA("Tool") then
                if item.Name == "Revolver" or item.Name == "Gun" then
                    return "Sheriff"
                elseif item.Name == "Knife" then
                    return "Murderer"
                end
            end
        end
    end
    
    if backpack then
        for _, item in ipairs(backpack:GetChildren()) do
            if item:IsA("Tool") then
                if item.Name == "Revolver" or item.Name == "Gun" then
                    return "Sheriff"
                elseif item.Name == "Knife" then
                    return "Murderer"
                end
            end
        end
    end
    
    return "Innocent"
end

local States = { InfiniteFling = false }
local InfiniteFlingTarget = nil
local OldPos = nil
local FPDH = workspace.FallenPartsDestroyHeight

local function StopFling()
    States.InfiniteFling = false
    InfiniteFlingTarget = nil
    local _, _, hrp = getChar()
    if hrp then
        hrp.AssemblyLinearVelocity = Vector3.new()
        hrp.AssemblyAngularVelocity = Vector3.new()
    end
end

local function SkidFling(TargetPlayer)
    local Character  = Player.Character
    local Humanoid   = Character and Character:FindFirstChildOfClass("Humanoid")
    local RootPart   = Humanoid and Humanoid.RootPart
    local TCharacter = TargetPlayer.Character
    if not TCharacter then return end

    local THumanoid = TCharacter:FindFirstChildOfClass("Humanoid")
    local TRootPart = THumanoid and THumanoid.RootPart
    local THead     = TCharacter:FindFirstChild("Head")
    local Accessory = TCharacter:FindFirstChildOfClass("Accessory")
    local Handle    = Accessory and Accessory:FindFirstChild("Handle")

    if not (Character and Humanoid and RootPart) then
        return Notify(tr("no_char"))
    end

    if RootPart.Velocity.Magnitude < 50 and not OldPos then OldPos = RootPart.CFrame end

    if THumanoid and THumanoid.Sit then
        return Notify(TargetPlayer.Name .. tr("sitting"))
    end

    workspace.CurrentCamera.CameraSubject = THead or Handle or THumanoid

    if not TCharacter:FindFirstChildWhichIsA("BasePart") then return end

    local function FPos(bp, pos, ang)
        RootPart.CFrame = CFrame.new(bp.Position) * pos * ang
        Character:SetPrimaryPartCFrame(CFrame.new(bp.Position) * pos * ang)
        RootPart.Velocity    = Vector3.new(9e7, 9e7*10, 9e7)
        RootPart.RotVelocity = Vector3.new(9e8, 9e8, 9e8)
    end

    local function SFBasePart(bp)
        local endT = tick() + 2
        local angle = 0
        repeat
            if RootPart and THumanoid then
                angle = angle + 100
                if bp.Velocity.Magnitude < 50 then
                    FPos(bp, CFrame.new(0,1.5,0) + THumanoid.MoveDirection*bp.Velocity.Magnitude/1.25, CFrame.Angles(math.rad(angle),0,0)) task.wait()
                    FPos(bp, CFrame.new(0,-1.5,0) + THumanoid.MoveDirection*bp.Velocity.Magnitude/1.25, CFrame.Angles(math.rad(angle),0,0)) task.wait()
                    FPos(bp, CFrame.new(0,1.5,0) + THumanoid.MoveDirection, CFrame.Angles(math.rad(angle),0,0)) task.wait()
                    FPos(bp, CFrame.new(0,-1.5,0) + THumanoid.MoveDirection, CFrame.Angles(math.rad(angle),0,0)) task.wait()
                else
                    FPos(bp, CFrame.new(0,1.5,THumanoid.WalkSpeed), CFrame.Angles(math.rad(90),0,0)) task.wait()
                    FPos(bp, CFrame.new(0,-1.5,-THumanoid.WalkSpeed), CFrame.Angles(0,0,0)) task.wait()
                    FPos(bp, CFrame.new(0,1.5,THumanoid.WalkSpeed), CFrame.Angles(math.rad(90),0,0)) task.wait()
                    FPos(bp, CFrame.new(0,-1.5,0), CFrame.Angles(math.rad(90),0,0)) task.wait()
                    FPos(bp, CFrame.new(0,-1.5,0), CFrame.Angles(0,0,0)) task.wait()
                end
            end
        until tick() > endT or not States.InfiniteFling
    end

    workspace.FallenPartsDestroyHeight = 0/0
    local BV = Instance.new("BodyVelocity")
    BV.Parent = RootPart; BV.Velocity = Vector3.new(); BV.MaxForce = Vector3.new(9e9,9e9,9e9)
    Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, false)

    local base = TRootPart or THead or Handle
    if base then SFBasePart(base)
    else BV:Destroy(); return Notify(TargetPlayer.Name .. tr("no_parts")) end

    BV:Destroy()
    Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
    workspace.CurrentCamera.CameraSubject = Humanoid

    if OldPos then
        repeat
            RootPart.CFrame = OldPos * CFrame.new(0,.5,0)
            Character:SetPrimaryPartCFrame(OldPos * CFrame.new(0,.5,0))
            Humanoid:ChangeState("GettingUp")
            for _, part in pairs(Character:GetChildren()) do
                if part:IsA("BasePart") then part.Velocity = Vector3.new(); part.RotVelocity = Vector3.new() end
            end
            task.wait()
        until (RootPart.Position - OldPos.p).Magnitude < 25
        workspace.FallenPartsDestroyHeight = FPDH
        OldPos = nil
    end
end

local function ExecuteFling(target)
    while States.InfiniteFling and InfiniteFlingTarget == target and target.Parent do
        SkidFling(target)
        task.wait(0.5)
    end
    StopFling()
end

local function Slider(page, labelKey, labelFallback, min, max, default, suffix, cb, isFloat)
    suffix = suffix or ""
    local row = Row(page, 46)
    local lbl = rowLabel(row, labelFallback or tr(labelKey))
    lbl.Position = UDim2.fromOffset(12,7); lbl.Size = UDim2.new(1,-90,0,16)
    if labelKey then onLang(function() lbl.Text = tr(labelKey) end) end

    local valLbl = new("TextLabel", { AnchorPoint = Vector2.new(1,0), Position = UDim2.new(1,-12,0,7),
        Size = UDim2.fromOffset(70,16), BackgroundTransparency = 1, Text = "",
        TextColor3 = Theme.Sub, Font = Enum.Font.GothamMedium, TextSize = 12,
        TextXAlignment = Enum.TextXAlignment.Right, Parent = row })
    local track = new("Frame", { Position = UDim2.new(0,12,1,-14), Size = UDim2.new(1,-24,0,4),
        BackgroundColor3 = Theme.Stroke, BorderSizePixel = 0, Parent = row })
    corner(track, 2)
    local fill = new("Frame", { Size = UDim2.fromScale(0,1), BackgroundColor3 = Theme.Accent,
        BorderSizePixel = 0, Parent = track })
    corner(fill, 2)
    local knob = new("Frame", { AnchorPoint = Vector2.new(0.5,0.5), Size = UDim2.fromOffset(10,10),
        Position = UDim2.fromScale(0,0.5), BackgroundColor3 = Color3.new(1,1,1),
        BorderSizePixel = 0, ZIndex = 2, Parent = track })
    circle(knob)
    onAccent(function(c) tween(fill, 0.25, { BackgroundColor3 = c }) end)

    local value = default
    local function render(v, animate)
        local pct = (v-min)/(max-min)
        if animate then
            tween(fill, 0.1, { Size = UDim2.fromScale(pct,1) }, Enum.EasingStyle.Quad)
            tween(knob, 0.1, { Position = UDim2.fromScale(pct,0.5) }, Enum.EasingStyle.Quad)
        else
            fill.Size = UDim2.fromScale(pct,1)
            knob.Position = UDim2.fromScale(pct,0.5)
        end
        if isFloat then
            valLbl.Text = string.format("%.1f", v) .. suffix
        else
            valLbl.Text = tostring(v) .. suffix
        end
    end
    render(default, false)

    local function setValue(v)
        if isFloat then
            v = math.clamp(math.floor(v * 10 + 0.5) / 10, min, max)
        else
            v = math.clamp(math.floor(v+0.5), min, max)
        end
        if v == value then return end
        value = v; render(v, true); cb(v)
    end

    local hit = new("TextButton", { Size = UDim2.fromScale(1,1), BackgroundTransparency = 1,
        Text = "", ZIndex = 3, Parent = row })
    local dragging = false
    local function fromX(x)
        local pct = math.clamp((x - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X,1), 0, 1)
        setValue(min + (max-min)*pct)
    end
    hit.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true; page.ScrollingEnabled = false; fromX(i.Position.X)
        end
    end)
    connect(UIS.InputChanged, function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            fromX(i.Position.X)
        end
    end)
    connect(UIS.InputEnded, function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch) then
            dragging = false; page.ScrollingEnabled = true
        end
    end)
end

local FarmState = { Active = false, Noclip = nil }

local function StopFarm()
    FarmState.Active = false
    if FarmState.Noclip then FarmState.Noclip:Disconnect(); FarmState.Noclip = nil end
    local _, hum, hrp = getChar()
    if hrp then
        local bv = hrp:FindFirstChild("FarmBV"); if bv then bv:Destroy() end
        local bg = hrp:FindFirstChild("FarmBG"); if bg then bg:Destroy() end
    end
    if hum then hum.PlatformStand = false end
end

local function GetMM2Coins()
    local coins = {}
    local container = workspace:FindFirstChild("Normal") and workspace.Normal:FindFirstChild("CoinContainer")
    local pool = container and container:GetChildren() or workspace:GetDescendants()
    
    for _, obj in ipairs(pool) do
        if obj:IsA("BasePart") and (obj.Name == "Coin_Server" or string.find(obj.Name, "Coin")) then
            if obj.Transparency < 1 then
                table.insert(coins, obj)
            end
        end
    end
    return coins
end

local function StartMM2Farm()
    if FarmState.Active then return end
    FarmState.Active = true

    task.spawn(function()
        FarmState.Noclip = RunService.Stepped:Connect(function()
            if Cfg.MM2Farm then
                local char = Player.Character
                if char then
                    for _, p in ipairs(char:GetDescendants()) do
                        if p:IsA("BasePart") then p.CanCollide = false end
                    end
                end
            end
        end)

        local coinRemote = ReplicatedStorage:FindFirstChild("CoinServer", true) or ReplicatedStorage:FindFirstChild("TransferCoin", true)

        while Cfg.MM2Farm and FarmState.Active do
            local char, hum, hrp = getChar()
            if not char or not hum or not hrp or hum.Health <= 0 then
                task.wait(1)
                continue
            end

            local coins = GetMM2Coins()
            if #coins == 0 then
                task.wait(1)
                continue
            end

            if coinRemote and coinRemote:IsA("RemoteEvent") then
                for _, coin in ipairs(coins) do
                    if not Cfg.MM2Farm or not FarmState.Active then break end
                    pcall(function()
                        coinRemote:FireServer(coin.Name)
                    end)
                    task.wait()
                end
            else
                local bestCoin, minDist = nil, math.huge
                for _, coin in ipairs(coins) do
                    local dist = (coin.Position - hrp.Position).Magnitude
                    if dist < minDist then
                        minDist = dist; bestCoin = coin
                    end
                end

                if bestCoin then
                    hum.PlatformStand = true

                    local bv = hrp:FindFirstChild("FarmBV") or Instance.new("BodyVelocity", hrp)
                    bv.Name = "FarmBV"; bv.MaxForce = Vector3.new(9e9, 9e9, 9e9); bv.Velocity = Vector3.new()

                    local bg = hrp:FindFirstChild("FarmBG") or Instance.new("BodyGyro", hrp)
                    bg.Name = "FarmBG"; bg.MaxTorque = Vector3.new(9e9, 9e9, 9e9); bg.CFrame = hrp.CFrame

                    local speed = 35
                    local duration = math.max(minDist / speed, 0.05)

                    local tw = TweenService:Create(hrp, TweenInfo.new(duration, Enum.EasingStyle.Linear), {
                        CFrame = bestCoin.CFrame + Vector3.new(0, 0.5, 0)
                    })
                    tw:Play()

                    local t = 0
                    while t < duration and Cfg.MM2Farm and FarmState.Active do
                        if not bestCoin or not bestCoin.Parent or bestCoin.Transparency == 1 then
                            break
                        end
                        if bg then bg.CFrame = hrp.CFrame end
                        t = t + task.wait()
                    end
                    tw:Cancel()

                    if bestCoin and bestCoin.Parent and bestCoin.Transparency < 1 then
                        pcall(function()
                            if firetouchinterest then
                                firetouchinterest(hrp, bestCoin, 0)
                                firetouchinterest(hrp, bestCoin, 1)
                            end
                        end)
                    end
                    task.wait()
                end
            end
        end
        StopFarm()
    end)
end

local function GrabGunAction()
    task.spawn(function()
        local _, _, hrp = getChar()
        if not hrp then return Notify(tr("no_char")) end
        local originPos = hrp.CFrame

        local gunDrop = nil
        for _, obj in ipairs(workspace:GetDescendants()) do
            if obj.Name == "GunDrop" and obj:IsA("BasePart") then
                gunDrop = obj
                break
            end
        end

        if gunDrop then
            hrp.CFrame = gunDrop.CFrame
            task.wait(0.2)
            pcall(function()
                if firetouchinterest then
                    firetouchinterest(hrp, gunDrop, 0)
                    firetouchinterest(hrp, gunDrop, 1)
                end
            end)
            task.wait(0.2)
            hrp.CFrame = originPos
            Notify("Arma pega e posição restaurada!")
            return
        end

        local sheriffPlayer = nil
        for _, pl in ipairs(Players:GetPlayers()) do
            if pl ~= Player then
                if getPlayerRole(pl) == "Sheriff" then
                    sheriffPlayer = pl
                    break
                end
            end
        end

        if sheriffPlayer then
            Notify("Xerife encontrado: " .. sheriffPlayer.Name .. ". Flingando...")
            OldPos = originPos
            States.InfiniteFling = true
            InfiniteFlingTarget = sheriffPlayer
            
            task.spawn(function()
                while States.InfiniteFling and InfiniteFlingTarget == sheriffPlayer and sheriffPlayer.Parent do
                    SkidFling(sheriffPlayer)
                    task.wait(0.5)
                end
            end)

            local timeout = tick() + 10
            while tick() < timeout do
                task.wait(0.3)
                for _, obj in ipairs(workspace:GetDescendants()) do
                    if obj.Name == "GunDrop" and obj:IsA("BasePart") then
                        gunDrop = obj
                        break
                    end
                end
                if gunDrop then break end
            end

            StopFling()

            if gunDrop then
                hrp.CFrame = gunDrop.CFrame
                task.wait(0.2)
                pcall(function()
                    if firetouchinterest then
                        firetouchinterest(hrp, gunDrop, 0)
                        firetouchinterest(hrp, gunDrop, 1)
                    end
                end)
                task.wait(0.2)
            end

            hrp.CFrame = originPos
            Notify("Ação concluída e posição restaurada!")
        else
            Notify("Nenhum Xerife com arma encontrado.")
        end
    end)
end

local function removeESP(pl)
    local e = ESP[pl]
    if e then e.gui:Destroy(); e.hl:Destroy(); ESP[pl] = nil end
end

local function updateESP()
    local _,_,myRoot = getChar()
    for pl, e in pairs(ESP) do
        local char  = pl.Character
        local head  = char and char:FindFirstChild("Head")
        local hum   = char and char:FindFirstChildOfClass("Humanoid")
        local alive = head and hum and hum.Health > 0

        if alive and (Cfg.ESPNames or Cfg.ESPHighlight) then
            local role = getPlayerRole(pl)
            local col = Color3.fromRGB(50, 255, 120)
            if role == "Murderer" then
                col = Color3.fromRGB(255, 50, 50)
            elseif role == "Sheriff" then
                col = Color3.fromRGB(50, 120, 255)
            end

            if Cfg.ESPNames then
                e.gui.Adornee = head; e.gui.Enabled = true
                e.name.Text = pl.DisplayName
                e.name.TextColor3 = col
                if Cfg.ESPDistance and myRoot then
                    e.dist.Text = math.floor((head.Position - myRoot.Position).Magnitude) .. " st"
                    e.dist.Visible = true
                else
                    e.dist.Visible = false
                end
            else
                e.gui.Enabled = false
            end
            if Cfg.ESPHighlight then
                e.hl.Adornee = char; e.hl.FillColor = col; e.hl.OutlineColor = col; e.hl.Enabled = true
            else
                e.hl.Enabled = false
            end
        else
            e.gui.Enabled = false; e.hl.Enabled = false
        end
    end
end

local rayParams = RaycastParams.new()
rayParams.FilterType = Enum.RaycastFilterType.Exclude

local function isVisible(part, char)
    local origin = Camera.CFrame.Position
    local filter = { char }
    if Player.Character then filter[2] = Player.Character end
    rayParams.FilterDescendantsInstances = filter
    return workspace:Raycast(origin, part.Position - origin, rayParams) == nil
end

local function getTarget()
    local vp = Camera.ViewportSize
    local center = Vector2.new(vp.X/2, vp.Y/2)
    local best, bestDist = nil, Cfg.AimFOV
    for _, pl in ipairs(Players:GetPlayers()) do
        if pl ~= Player then
            local char = pl.Character
            local hum  = char and char:FindFirstChildOfClass("Humanoid")
            local part = char and (char:FindFirstChild(Cfg.AimPart) or char:FindFirstChild("HumanoidRootPart"))
            local sameTeam = Cfg.AimTeam and pl.Team ~= nil and pl.Team == Player.Team
            if part and hum and hum.Health > 0 and not sameTeam then
                local v, onScreen = Camera:WorldToViewportPoint(part.Position)
                if onScreen then
                    local d = (Vector2.new(v.X, v.Y) - center).Magnitude
                    if d < bestDist
                        and (part.Position - Camera.CFrame.Position).Magnitude <= Cfg.AimDist
                        and (not Cfg.AimWall or isVisible(part, char)) then
                        best, bestDist = part, d
                    end
                end
            end
        end
    end
    return best
end

RunService:BindToRenderStep("BigHubAim", Enum.RenderPriority.Camera.Value + 1, function(dt)
    Camera = workspace.CurrentCamera
    FovCircle.Visible = Cfg.Aim and Cfg.AimShowFOV
    if not Cfg.Aim then return end
    if Cfg.AimHold and not UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then return end
    local target = getTarget()
    if not target then return end
    local cam  = Camera.CFrame
    local goal = CFrame.lookAt(cam.Position, target.Position)
    local s    = math.clamp(Cfg.AimSmooth / 100, 0, 0.98)
    local alpha = 1 - s ^ (dt * 60)
    Camera.CFrame = cam:Lerp(goal, alpha)
end)

connect(RunService.RenderStepped, function()
    updateESP()

    if Cfg.Fly then
        local _, hum, hrp = getChar()
        if hum and hrp then
            local bv = hrp:FindFirstChild("BHFlyVel")
            local bg = hrp:FindFirstChild("BHFlyGyro")
            if not bv then
                bv = new("BodyVelocity", { Name = "BHFlyVel", MaxForce = Vector3.new(9e9,9e9,9e9), Velocity = Vector3.new(), Parent = hrp })
            end
            if not bg then
                bg = new("BodyGyro", { Name = "BHFlyGyro", MaxTorque = Vector3.new(9e9,9e9,9e9), P = 9e4, Parent = hrp })
            end
            local cf = workspace.CurrentCamera.CFrame
            bg.CFrame = cf
            local move       = hum.MoveDirection
            local flatLook   = Vector3.new(cf.LookVector.X,  0, cf.LookVector.Z)
            local flatRight  = Vector3.new(cf.RightVector.X, 0, cf.RightVector.Z)
            local fwd, rgt   = 0, 0
            if flatLook.Magnitude  > 0.001 then fwd = move:Dot(flatLook.Unit)  end
            if flatRight.Magnitude > 0.001 then rgt = move:Dot(flatRight.Unit) end
            local vel = cf.LookVector * fwd + cf.RightVector * rgt
            if UIS:IsKeyDown(Enum.KeyCode.Space)        then vel = vel + Vector3.new(0,1,0) end
            if UIS:IsKeyDown(Enum.KeyCode.LeftControl) or UIS:IsKeyDown(Enum.KeyCode.LeftShift) then
                vel = vel - Vector3.new(0,1,0)
            end
            bv.Velocity = vel * Cfg.FlySpeed
        end
    end
end)
