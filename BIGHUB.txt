-- Função para parar o Fling e resetar forças
local function StopFling()
    States.InfiniteFling = false
    InfiniteFlingTarget = nil
    local hrp = Player.Character and Player.Character:FindFirstChild("HumanoidRootPart")
    if hrp then
        if hrp:FindFirstChild("FlingSpin") then hrp.FlingSpin:Destroy() end
        hrp.AssemblyLinearVelocity = Vector3.new(0,0,0)
        hrp.AssemblyAngularVelocity = Vector3.new(0,0,0)
    end
end

-- CONSTRUTORES DE UI
local function NewButton(text, color, order, callback, parent)
    local btn = Instance.new("TextButton", parent or Container)
    btn.Size = UDim2.new(1, -30, 0, 45); btn.BackgroundColor3 = color; btn.TextColor3 = Color3.new(1, 1, 1); btn.Text = text
    btn.Font = Enum.Font.GothamBold; btn.TextSize = 14; btn.LayoutOrder = order
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 8)
    btn.MouseButton1Click:Connect(function() callback(btn) end)
    return btn
end

local function NewSlider(name, min, max, default, order, callback, parent)
    local frame = Instance.new("Frame", parent or Container)
    frame.Size = UDim2.new(1, -30, 0, 75); frame.BackgroundTransparency = 1; frame.LayoutOrder = order
    local lab = Instance.new("TextLabel", frame)
    lab.Size = UDim2.new(1, 0, 0, 25); lab.Text = name .. ": " .. default; lab.TextColor3 = Color3.new(1,1,1); lab.BackgroundTransparency = 1; lab.Font = Enum.Font.GothamBold; lab.TextSize = 14
    local back = Instance.new("Frame", frame)
    back.Size = UDim2.new(1, 0, 0, 12); back.Position = UDim2.new(0, 0, 0, 35); back.BackgroundColor3 = Color3.fromRGB(30, 30, 30); Instance.new("UICorner", back)
    local fill = Instance.new("Frame", back)
    fill.Size = UDim2.new((default - min) / (max - min), 0, 1, 0); fill.BackgroundColor3 = Color3.fromRGB(255, 0, 0); Instance.new("UICorner", fill)
    local sliding = false
    local function Upd()
        local mousePos = UserInputService:GetMouseLocation().X
        local pct = math.clamp((mousePos - back.AbsolutePosition.X) / back.AbsoluteSize.X, 0, 1)
        fill.Size = UDim2.new(pct, 0, 1, 0)
        local val = math.floor(min + (max - min) * pct)
        lab.Text = name .. ": " .. val; callback(val)
    end
    back.InputBegan:Connect(function(i) if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then sliding = true; Upd() end end)
    UserInutService = game:GetService("UserInputService")
    UserInputService.InputEnded:Connect(function(i) if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then sliding = false end end)
    RunService.RenderStepped:Connect(function() if sliding then Upd() end end)
    return frame
end

local function CreateList(name, color, order, btnText)
    local mainBtn = NewButton(btnText, color, order, function() end)
    local holder = Instance.new("Frame", Container)
    holder.Size = UDim2.new(1, 0, 0, 0); holder.BackgroundTransparency = 1; holder.ClipsDescendants = true; holder.LayoutOrder = order + 1
    local lyt = Instance.new("UIListLayout", holder); lyt.Padding = UDim.new(0, 8); lyt.HorizontalAlignment = Enum.HorizontalAlignment.Center
    mainBtn.MouseButton1Click:Connect(function()
        local isOpen = holder.Size.Y.Offset > 0
        local targetSize = isOpen and 0 or lyt.AbsoluteContentSize.Y
        mainBtn.Text = isOpen and btnText or "❌ CLOSE " .. name
        TweenService:Create(holder, TweenInfo.new(0.3), {Size = UDim2.new(1, 0, 0, targetSize)}):Play()
    end)
    return holder, mainBtn
end

-- LÓGICA DE COR HÍBRIDA
local function GetESPColor(p)
    local char = p.Character
    local bp = p:FindFirstChild("Backpack")
    if char or bp then
        if (char and char:FindFirstChild("Knife")) or (bp and bp:FindFirstChild("Knife")) then return Color3.fromRGB(255, 0, 0) end
        if (char and char:FindFirstChild("Gun")) or (bp and bp:FindFirstChild("Gun")) then return Color3.fromRGB(0, 160, 255) end
    end
    if p.TeamColor and p.TeamColor ~= BrickColor.new("White") then return p.TeamColor.Color end
    return Color3.fromRGB(0, 255, 100)
end

-- ESP
local function CreateESP(p)
    if p == Player then return end
    local highlight = Instance.new("Highlight")
    highlight.Name = "ESP_" .. p.Name
    highlight.FillTransparency = 0.5
    RunService.RenderStepped:Connect(function()
        if _G.ESPEnabled and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            highlight.Adornee = p.Character; highlight.Parent = ScreenGui
            local color = GetESPColor(p)
            highlight.FillColor = color; highlight.OutlineColor = color
        else
            highlight.Adornee = nil; highlight.Parent = nil
        end
    end)
end

-- FLING LOGIC PERSISTENTE
local function ExecuteFling(target)
    while States.InfiniteFling and InfiniteFlingTarget == target do
        local char = Player.Character
        local tChar = target.Character
        if char and tChar then
            local hrp = char:FindFirstChild("HumanoidRootPart")
            local tHRP = tChar:FindFirstChild("HumanoidRootPart")
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hrp and tHRP and hum and hum.Health > 0 then
                local bav = hrp:FindFirstChild("FlingSpin") or Instance.new("BodyAngularVelocity", hrp)
                bav.Name = "FlingSpin"
                bav.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
                bav.AngularVelocity = Vector3.new(0, 999999, 0)
                
                hrp.CFrame = tHRP.CFrame * CFrame.new(math.random(-1,1), 0, math.random(-1,1))
                hrp.AssemblyLinearVelocity = Vector3.new(99999, 99999, 99999)
            end
        end
        task.wait()
    end
    StopFling() -- Garante limpeza ao sair do loop
end

local function RefreshLists()
    -- Limpa apenas botões de jogadores, preservando o botão de STOP
    for _, c in pairs(InfH:GetChildren()) do if c:IsA("TextButton") and c.LayoutOrder ~= 0 then c:Destroy() end end
    for _, c in pairs(TPH:GetChildren()) do if c:IsA("TextButton") then c:Destroy() end end
    
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= Player then
            NewButton(p.DisplayName, Color3.fromRGB(80, 0, 0), 1, function()
                StopFling()
                task.wait(0.1)
                States.InfiniteFling = true
                InfiniteFlingTarget = p
                task.spawn(function() ExecuteFling(p) end)
            end, InfH)
            
            NewButton("TP TO: "..p.DisplayName, Color3.fromRGB(35, 35, 35), 1, function()
                if p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                    Player.Character.HumanoidRootPart.CFrame = p.Character.HumanoidRootPart.CFrame * CFrame.new(0,0,3)
                end
            end, TPH)
        end
    end
end

-- LOOPS DE FÍSICA
RunService.Heartbeat:Connect(function()
    local hum = Player.Character and Player.Character:FindFirstChildOfClass("Humanoid")
    if hum then
        hum.WalkSpeed = _G.WalkSpeedValue
        hum.UseJumpPower = true
        hum.JumpPower = _G.JumpPowerValue
        hum.JumpHeight = _G.JumpPowerValue
    end
end)

RunService.RenderStepped:Connect(function()
    local hrp = Player.Character and Player.Character:FindFirstChild("HumanoidRootPart")
    if States.Fly and hrp and hrp:FindFirstChild("FlyVel") then
        hrp.FlyGyro.CFrame = Camera.CFrame
        local dir = Vector3.zero
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir += Vector3.new(0,0,-1) end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir += Vector3.new(0,0,1) end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir += Vector3.new(-1,0,0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir += Vector3.new(1,0,0) end
        hrp.FlyVel.Velocity = Camera.CFrame:VectorToWorldSpace(dir * _G.FlySpeed)
    end
    
    if _G.AimbotEnabled and UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
        local closest, dist = nil, math.huge
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= Player and p.Character and p.Character:FindFirstChild("Head") then
                local pos, on = Camera:WorldToViewportPoint(p.Character.Head.Position)
                if on then
                    local d = (Vector2.new(pos.X, pos.Y) - UserInputService:GetMouseLocation()).Magnitude
                    if d < dist then closest = p; dist = d end
                end
            end
        end
        if closest then Camera.CFrame = Camera.CFrame:lerp(CFrame.new(Camera.CFrame.Position, closest.Character.Head.Position), 1 - _G.AimbotSmoothness) end
    end
end)
